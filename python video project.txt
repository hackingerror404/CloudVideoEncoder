import os
import threading
import subprocess
import boto3
import flet as ft

def run_ffmpeg_encode(input_path, output_path, video_codec, crf):
    cmd = [
        "ffmpeg",
        "-y",
        "-i", input_path,
        "-c:v", video_codec,
        "-crf", str(crf),
        "-c:a", "copy",
        output_path
    ]
    proc = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    return proc.returncode == 0, proc.stdout, proc.stderr

def scan_videos(input_directory, allowed_exts=None):
    if allowed_exts is None:
        allowed_exts = {".mp4", ".mov", ".mkv", ".avi"}
    videos = []
    for root, _, files in os.walk(input_directory):
        for f in files:
            ext = os.path.splitext(f)[1].lower()
            if ext in allowed_exts:
                videos.append(os.path.join(root, f))
    return sorted(videos)

def main(page: ft.Page):
    page.title = "Encode & Upload with Progress"
    page.window_width = 700
    page.window_height = 420
    page.padding = 16

    input_dir = ft.TextField(label="Local Video Path", width=600)
    output_prefix = ft.TextField(label="Cloud Output Prefix (e.g. vidsOutput/)", width=600)
    start_btn = ft.ElevatedButton(text="Start", width=160)
    status_label = ft.Text("", size=14)
    progress_bar = ft.ProgressBar(width=600, height=12, bgcolor=ft.colors.GREY_200)
    progress_text = ft.Text("0 / 0", size=12)

    # state
    progress_state = {"done": 0, "total": 0}

    def set_status(text: str):
        status_label.value = text
        page.update()

    def set_progress(done: int, total: int):
        progress_state["done"] = done
        progress_state["total"] = total
        frac = (done / total) if total > 0 else 0.0
        progress_bar.value = frac
        progress_text.value = f"{done} / {total}"
        page.update()

    s3 = boto3.client("s3")

    def worker(input_path: str, output_prefix_value: str):
        videos = scan_videos(input_path)
        num = len(videos)
        if num == 0:
            page.call_from_thread(lambda: set_status("No videos found."))
            return

        total_tasks = num * 2
        done = 0
        page.call_from_thread(lambda: set_progress(done, total_tasks))

        for idx, vid_path in enumerate(videos, start=1):
            base = os.path.splitext(os.path.basename(vid_path))[0]
            temp_output = os.path.join(input_path, f"{base}_temp.mp4")
            aws_key = (output_prefix_value.rstrip("/") + "/" + base + ".mp4").lstrip("/")

            # Encoding step
            page.call_from_thread(lambda v=base, i=idx, n=num: set_status(f"Encoding {v} ({i}/{n})"))
            ok, _, _ = run_ffmpeg_encode(vid_path, temp_output, "libx264", 23)
            done += 1
            page.call_from_thread(lambda d=done, t=total_tasks: set_progress(d, t))

            if not ok:
                page.call_from_thread(lambda v=base: set_status(f"Encoding failed: {v} — skipping upload"))
                continue

            # Upload step
            page.call_from_thread(lambda v=base, i=idx, n=num: set_status(f"Uploading {v} ({i}/{n})"))
            try:
                s3.upload_file(temp_output, "hackingerror404-bucket", aws_key)
                page.call_from_thread(lambda v=base: set_status(f"Uploaded {v}"))
            except Exception as e:
                page.call_from_thread(lambda e=e, v=base: set_status(f"Upload failed for {v}: {e}"))
            finally:
                if os.path.exists(temp_output):
                    try:
                        os.remove(temp_output)
                    except Exception:
                        pass

            done += 1
            page.call_from_thread(lambda d=done, t=total_tasks: set_progress(d, t))

        page.call_from_thread(lambda: set_status("All tasks complete."))

    def on_start(e):
        if not input_dir.value:
            set_status("Please provide a local video path.")
            return
        if not output_prefix.value:
            set_status("Please provide a cloud output prefix.")
            return
        # reset UI
        set_status("Preparing...")
        set_progress(0, 1)  # temporary while scanning
        threading.Thread(target=worker, args=(input_dir.value, output_prefix.value), daemon=True).start()

    start_btn.on_click = on_start

    page.add(
        ft.Column([
            input_dir,
            output_prefix,
            ft.Row([start_btn, ft.Container(width=12), progress_text]),
            ft.Container(height=8),
            progress_bar,
            ft.Container(height=12),
            status_label
        ])
    )

ft.app(target=main)
